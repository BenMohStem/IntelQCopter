cmake_minimum_required(VERSION 3.16)
project(qcopter_autonomy)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc dnn calib3d)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/AutonomySetpoint.msg"
  "msg/SemanticDetection.msg" "msg/SemanticDetections.msg" "msg/MetricObstacles.msg"
  "srv/RequestMission.srv" "srv/RequestRoute.srv" DEPENDENCIES std_msgs geometry_msgs)
add_library(navigation_core STATIC src/navigation.cpp src/mission.cpp)
set_target_properties(navigation_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(navigation_core PUBLIC include)
add_executable(navigator_node src/navigator_node.cpp)
target_link_libraries(navigator_node navigation_core)
ament_target_dependencies(navigator_node rclcpp geometry_msgs nav_msgs sensor_msgs std_msgs)
rosidl_get_typesupport_target(typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(navigator_node "${typesupport_target}")
add_executable(yolo_node src/yolo_node.cpp)
target_include_directories(yolo_node PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(yolo_node ${OpenCV_LIBRARIES} "${typesupport_target}")
ament_target_dependencies(yolo_node rclcpp sensor_msgs std_msgs)
install(TARGETS navigator_node DESTINATION lib/${PROJECT_NAME})
install(TARGETS yolo_node DESTINATION lib/${PROJECT_NAME})
add_executable(mission_node src/mission_node.cpp)
target_link_libraries(mission_node navigation_core "${typesupport_target}")
ament_target_dependencies(mission_node rclcpp nav_msgs geometry_msgs std_msgs std_srvs)
add_executable(mission_client src/mission_client.cpp)
target_link_libraries(mission_client "${typesupport_target}")
ament_target_dependencies(mission_client rclcpp geometry_msgs)
install(TARGETS mission_node mission_client DESTINATION lib/${PROJECT_NAME})
add_executable(panel_pose_node src/panel_pose_node.cpp)
target_include_directories(panel_pose_node PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(panel_pose_node ${OpenCV_LIBRARIES} "${typesupport_target}")
ament_target_dependencies(panel_pose_node rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs)
install(TARGETS panel_pose_node DESTINATION lib/${PROJECT_NAME})
add_executable(pilot_mission_node src/pilot_mission_node.cpp)
target_link_libraries(pilot_mission_node navigation_core "${typesupport_target}")
ament_target_dependencies(pilot_mission_node rclcpp nav_msgs std_msgs)
install(TARGETS pilot_mission_node DESTINATION lib/${PROJECT_NAME})
add_executable(uart_bridge_node src/uart_bridge_node.cpp)
ament_target_dependencies(uart_bridge_node rclcpp std_msgs)
install(TARGETS uart_bridge_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
